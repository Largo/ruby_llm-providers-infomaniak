infomaniak.rb
require 'ruby_llm/providers/infomaniak'
RubyLLM.configure { |c| c.infomaniak_api_key = ENV['INFOMANIAK_API_KEY'] }
 
chat = RubyLLM.chat(model: 'moonshotai/Kimi-K2.6', provider: :infomaniak)
chat.ask('Grüezi! Why do people love Ruby?') { |chunk| print chunk.content }
 
chat.with_tools(Weather).ask('Do I need an umbrella in Zurich?')
chat.with_schema(Receipt).ask('Extract the total', with: 'receipt.jpg').parsed
RubyLLM.paint('A red train on a viaduct', model: 'flux', provider: :infomaniak)
 
chat.cost.total # => 0.0045 (CHF)
